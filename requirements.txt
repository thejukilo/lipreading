# Phase 1 inference-server dependencies.
#
# IMPORTANT (Windows + CUDA): install torch/torchvision/torchaudio from the
# CUDA wheel index FIRST, otherwise pip pulls the CPU-only build:
#
#   pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu121
#
# Then: pip install -r requirements.txt
#
# auto_avsr itself is NOT installed via pip — it is cloned into
# third_party/auto_avsr and put on sys.path by server/engine.py. Its vendored
# espnet decoder and sentencepiece tokenizer live inside that checkout.

# mediapipe: any modern version works. We use its Tasks FaceDetector API
# (server/detectors.py) because the legacy mp.solutions API auto_avsr relied on
# was removed in mediapipe >= ~0.10.18.
torch
torchvision
torchaudio
pytorch-lightning
sentencepiece
av
mediapipe
opencv-python
scikit-image
numpy
gdown        # downloads the better checkpoints from Google Drive (large-file token)

# Live app (step 2+): webcam capture, push-to-talk, TTS, virtual-mic playback
pynput       # global push-to-talk hotkey (no admin needed)
sounddevice  # play synthesized audio into the VB-Cable virtual mic
soundfile    # read WAV for playback
piper-tts    # local neural TTS (default voice engine)
pyttsx3      # Windows SAPI5 fallback voice (no model download)
PySide6      # desktop setup UI (server.gui_app)
